Cara menjalankan Relay Agent

Relay Agent adalah program kecil yang dijalankan di mesin lokal Anda. Ia menjembatani Relay Server dengan service lokal Anda tanpa perlu IP publik.

Download binary sesuai OS Anda — tidak perlu install apapun. 

# 1. Download binary (Apple Silicon / Intel)
# Apple Silicon — M1/M2/M3/M4
curl -L "https://github.com/theKims0/relay-agent/releases/latest/download/relay-agent-darwin-arm64" -o relay-agent

# Intel Mac
curl -L "https://github.com/theKims0/relay-agent/releases/latest/download/relay-agent-darwin-amd64" -o relay-agent

# 2. Beri izin eksekusi
chmod +x relay-agent

# 3. Jalankan — wizard otomatis akan muncul saat pertama kali
./relay-agent

Cara mengirim request via Relay

Kirim request HTTP biasa ke Relay Server dengan menambahkan header X-Relay-Token. Server akan meneruskannya ke service lokal Anda secara otomatis.

Base URL: http://localhost:8081/v1/gateway/slug 
# GET request
curl -X GET http://localhost:8081/v1/gateway/slug/api/users \
  -H "X-Relay-Token: YOUR_RELAY_TOKEN"

# POST request dengan JSON body
curl -X POST http://localhost:8081/v1/gateway/slug/api/users \
  -H "X-Relay-Token: YOUR_RELAY_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"name": "John", "email": "john@example.com"}'

# PUT request
curl -X PUT http://localhost:8081/v1/gateway/slug/api/users/1 \
  -H "X-Relay-Token: YOUR_RELAY_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"name": "John Updated"}'
// Konfigurasi
const RELAY_BASE = 'http://localhost:8081/v1/gateway/slug';
const RELAY_TOKEN = 'YOUR_RELAY_TOKEN';

// Helper function
async function relayFetch(path, options = {}) {
  return fetch(`${RELAY_BASE}${path}`, {
    ...options,
    headers: {
      'X-Relay-Token': RELAY_TOKEN,
      'Content-Type': 'application/json',
      ...options.headers,
    },
  });
}

// GET
const users = await relayFetch('/api/users');
const data  = await users.json();

// POST
const res = await relayFetch('/api/users', {
  method: 'POST',
  body: JSON.stringify({ name: 'John', email: 'john@example.com' }),
});

package main

import (
    "bytes"
    "encoding/json"
    "fmt"
    "net/http"
)

const (
    relayBase  = "http://localhost:8081/v1/gateway/slug"
    relayToken = "YOUR_RELAY_TOKEN"
)

func relayDo(method, path string, body any) (*http.Response, error) {
    var buf bytes.Buffer
    if body != nil {
        json.NewEncoder(&buf).Encode(body)
    }
    req, _ := http.NewRequest(method, relayBase+path, &buf)
    req.Header.Set("X-Relay-Token", relayToken)
    req.Header.Set("Content-Type", "application/json")
    return http.DefaultClient.Do(req)
}

func main() {
    // GET
    resp, _ := relayDo("GET", "/api/users", nil)
    fmt.Println(resp.Status)

    // POST
    resp, _ = relayDo("POST", "/api/users", map[string]string{
        "name": "John", "email": "john@example.com",
    })
    fmt.Println(resp.Status)
}
import java.net.URI;
import java.net.http.*;

String relayBase  = "http://localhost:8081/v1/gateway/slug";
String relayToken = "YOUR_RELAY_TOKEN";

HttpClient client = HttpClient.newHttpClient();

// GET
HttpRequest getReq = HttpRequest.newBuilder()
    .uri(URI.create(relayBase + "/api/users"))
    .header("X-Relay-Token", relayToken)
    .GET()
    .build();
HttpResponse<String> getResp = client.send(getReq, HttpResponse.BodyHandlers.ofString());
System.out.println(getResp.body());

// POST
String body = """{"name":"John","email":"john@example.com"}""";
HttpRequest postReq = HttpRequest.newBuilder()
    .uri(URI.create(relayBase + "/api/users"))
    .header("X-Relay-Token", relayToken)
    .header("Content-Type", "application/json")
    .POST(HttpRequest.BodyPublishers.ofString(body))
    .build();
HttpResponse<String> postResp = client.send(postReq, HttpResponse.BodyHandlers.ofString());
System.out.println(postResp.statusCode());
<?php
$relayBase  = 'http://localhost:8081/v1/gateway/slug';
$relayToken = 'YOUR_RELAY_TOKEN';

// Helper function
function relayRequest($method, $path, $data = null) {
    global $relayBase, $relayToken;
    $ch = curl_init($relayBase . $path);
    $headers = ["X-Relay-Token: $relayToken", "Content-Type: application/json"];
    curl_setopt_array($ch, [
        CURLOPT_CUSTOMREQUEST  => $method,
        CURLOPT_HTTPHEADER     => $headers,
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_POSTFIELDS     => $data ? json_encode($data) : null,
    ]);
    $res = curl_exec($ch);
    curl_close($ch);
    return json_decode($res, true);
}

// GET
$users = relayRequest('GET', '/api/users');
var_dump($users);

// POST
$result = relayRequest('POST', '/api/users', [
    'name'  => 'John',
    'email' => 'john@example.com',
]);
var_dump($result);
?>
